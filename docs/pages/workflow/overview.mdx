import React, { useState } from "react";
import { NavigationContainer } from "@react-navigation/native";
import { createStackNavigator } from "@react-navigation/stack";
import { View, Text, TextInput, Button, FlatList, TouchableOpacity } from "react-native";

const Stack = createStackNavigator();

function HomeScreen({ navigation }) {
  return (
    <View style={{ flex: 1, justifyContent: "center", alignItems: "center" }}>
      <Text style={{ fontSize: 22, marginBottom: 20 }}>Ensino Domiciliar</Text>
      <Button title="Registro de Aula" onPress={() => navigation.navigate("Registro de Aula")} />
      <Button title="Calendário" onPress={() => navigation.navigate("Calendário")} />
      <Button title="Filhos" onPress={() => navigation.navigate("Filhos")} />
      <Button title="Relatórios" onPress={() => navigation.navigate("Relatórios")} />
    </View>
  );
}

function RegistroScreen({ route }) {
  const [data, setData] = useState("");
  const [materia, setMateria] = useState("");
  const [atividade, setAtividade] = useState("");
  const [observacao, setObservacao] = useState("");
  const [registros, setRegistros] = useState([]);

  const salvarRegistro = () => {
    const novo = { data, materia, atividade, observacao, id: Date.now().toString() };
    setRegistros([...registros, novo]);
    setData(""); setMateria(""); setAtividade(""); setObservacao("");
  };

  return (
    <View style={{ flex: 1, padding: 20 }}>
      <Text>Data:</Text>
      <TextInput value={data} onChangeText={setData} placeholder="DD/MM/AAAA" style={{ borderWidth: 1, marginBottom: 10, padding: 5 }} />
      <Text>Matéria:</Text>
      <TextInput value={materia} onChangeText={setMateria} placeholder="Ex: Matemática" style={{ borderWidth: 1, marginBottom: 10, padding: 5 }} />
      <Text>Atividade:</Text>
      <TextInput value={atividade} onChangeText={setAtividade} placeholder="Ex: Frações" style={{ borderWidth: 1, marginBottom: 10, padding: 5 }} />
      <Text>Observações:</Text>
      <TextInput value={observacao} onChangeText={setObservacao} placeholder="Ex: Gostou bastante" style={{ borderWidth: 1, marginBottom: 10, padding: 5 }} />
      <Button title="Salvar Registro" onPress={salvarRegistro} />

      <FlatList
        data={registros}
        keyExtractor={(item) => item.id}
        renderItem={({ item }) => (
          <Text style={{ marginTop: 10 }}>{item.data} - {item.materia}: {item.atividade}</Text>
        )}
      />
    </View>
  );
}

function CalendarioScreen() {
  return (
    <View style={{ flex: 1, justifyContent: "center", alignItems: "center" }}>
      <Text>📅 Aqui você poderá visualizar os registros por data.</Text>
    </View>
  );
}

function FilhosScreen() {
  const [nome, setNome] = useState("");
  const [idade, setIdade] = useState("");
  const [filhos, setFilhos] = useState([]);

  const salvarFilho = () => {
    const novo = { nome, idade, id: Date.now().toString() };
    setFilhos([...filhos, novo]);
    setNome(""); setIdade("");
  };

  return (
    <View style={{ flex: 1, padding: 20 }}>
      <Text>Nome:</Text>
      <TextInput value={nome} onChangeText={setNome} placeholder="Nome da criança" style={{ borderWidth: 1, marginBottom: 10, padding: 5 }} />
      <Text>Idade:</Text>
      <TextInput value={idade} onChangeText={setIdade} placeholder="Idade" style={{ borderWidth: 1, marginBottom: 10, padding: 5 }} />
      <Button title="Salvar Filho" onPress={salvarFilho} />

      <FlatList
        data={filhos}
        keyExtractor={(item) => item.id}
        renderItem={({ item }) => (
          <Text style={{ marginTop: 10 }}>{item.nome} - {item.idade} anos</Text>
        )}
      />
    </View>
  );
}

function RelatoriosScreen() {
  return (
    <View style={{ flex: 1, justifyContent: "center", alignItems: "center" }}>
      <Text>📊 Aqui você verá relatórios e gráficos futuramente.</Text>
    </View>
  );
}

export default function App() {
  return (
    <NavigationContainer>
      <Stack.Navigator initialRouteName="Home">
        <Stack.Screen name="Início" component={HomeScreen} />
        <Stack.Screen name="Registro de Aula" component={RegistroScreen} />
        <Stack.Screen name="Calendário" component={CalendarioScreen} />
        <Stack.Screen name="Filhos" component={FilhosScreen} />
        <Stack.Screen name="Relatórios" component={RelatoriosScreen} />
      </Stack.Navigator>
    </NavigationContainer>
  );
}
